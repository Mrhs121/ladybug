# The graph implementation lives in liblbug. DuckDB only reads the Iceberg snapshots.
if(WIN32)
    set(iceberg_runtime_dir "${CMAKE_INSTALL_BINDIR}")
else()
    set(iceberg_runtime_dir "${CMAKE_INSTALL_LIBDIR}")
endif()

add_library(iceberg_duckdb INTERFACE)
if(DuckDB_FOUND AND NOT DUCKDB_DIR)
    target_link_libraries(iceberg_duckdb INTERFACE ${DuckDB_LIBRARIES})
    target_include_directories(iceberg_duckdb SYSTEM INTERFACE ${DuckDB_INCLUDE_DIRS})
    set(duckdb_runtime "$<TARGET_FILE:${DuckDB_LIBRARIES}>")
else()
    if(NOT DUCKDB_DIR)
        include(FetchContent)
        if(CMAKE_SYSTEM_NAME STREQUAL "Darwin")
            set(sdk osx-universal)
            set(sha 7b5b8915cc382d0708636fe6385c0cdad5a61c9ff8ba2638b3e2141640783155)
        elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64|aarch64|arm64)$")
            file(GLOB musl_loader "${CMAKE_SYSROOT}/lib/ld-musl-*.so.1")
            if(musl_loader AND OS_ARCH STREQUAL "arm64")
                set(sdk linux-arm64-musl)
                set(sha 800fc0217600d10cd4a74215a19df5e998393877f2a91e2b998c1cabdf75ba80)
            elseif(musl_loader)
                set(sdk linux-amd64-musl)
                set(sha 424f275a84cd488eb65329d4b33265f9fae93212396ab40c8048fc2f3f07817f)
            elseif(OS_ARCH STREQUAL "arm64")
                set(sdk linux-arm64)
                set(sha abe4f6f005ee0b448a058322f4263584b4bd1b6faf7ab4637b79eeaf978f8e9c)
            else()
                set(sdk linux-amd64)
                set(sha 1fb8ce388157d84a25abe685a8a2520bf00c00321821968e4bb398fd766e7abb)
            endif()
        else()
            message(FATAL_ERROR "Set DUCKDB_DIR to a DuckDB SDK for this target")
        endif()
        FetchContent_Declare(iceberg_duckdb_sdk
            URL "https://github.com/duckdb/duckdb/releases/download/v1.5.5/libduckdb-${sdk}.zip"
            URL_HASH SHA256=${sha})
        FetchContent_MakeAvailable(iceberg_duckdb_sdk)
        set(DUCKDB_DIR "${iceberg_duckdb_sdk_SOURCE_DIR}")
    endif()
    unset(ICEBERG_DUCKDB_INCLUDE CACHE)
    unset(ICEBERG_DUCKDB_LIBRARY CACHE)
    find_path(ICEBERG_DUCKDB_INCLUDE duckdb.hpp PATHS "${DUCKDB_DIR}" "${DUCKDB_DIR}/include" NO_DEFAULT_PATH)
    find_file(ICEBERG_DUCKDB_LIBRARY NAMES "${CMAKE_SHARED_LIBRARY_PREFIX}duckdb${CMAKE_SHARED_LIBRARY_SUFFIX}"
        PATHS "${DUCKDB_DIR}" "${DUCKDB_DIR}/lib" "${DUCKDB_DIR}/bin" NO_DEFAULT_PATH)
    if(NOT ICEBERG_DUCKDB_INCLUDE OR NOT ICEBERG_DUCKDB_LIBRARY)
        message(FATAL_ERROR "Missing DuckDB headers or shared library in ${DUCKDB_DIR}")
    endif()
    add_library(iceberg_duckdb_sdk_lib SHARED IMPORTED GLOBAL)
    set_target_properties(iceberg_duckdb_sdk_lib PROPERTIES IMPORTED_LOCATION "${ICEBERG_DUCKDB_LIBRARY}"
        INTERFACE_INCLUDE_DIRECTORIES "${ICEBERG_DUCKDB_INCLUDE}")
    if(WIN32)
        unset(ICEBERG_DUCKDB_IMPLIB CACHE)
        find_library(ICEBERG_DUCKDB_IMPLIB NAMES duckdb PATHS "${DUCKDB_DIR}" "${DUCKDB_DIR}/lib" NO_DEFAULT_PATH)
        set_property(TARGET iceberg_duckdb_sdk_lib PROPERTY IMPORTED_IMPLIB "${ICEBERG_DUCKDB_IMPLIB}")
    elseif(NOT APPLE)
        set_property(TARGET iceberg_duckdb_sdk_lib PROPERTY IMPORTED_NO_SONAME TRUE)
    endif()
    target_link_libraries(iceberg_duckdb INTERFACE iceberg_duckdb_sdk_lib)
    set(duckdb_runtime "${ICEBERG_DUCKDB_LIBRARY}")
endif()
find_package(SQLite3 REQUIRED)
# The macOS compiler already searches the system SDK headers. Adding usr/include
# explicitly would place C headers ahead of the C++ standard library.
if(APPLE AND SQLite3_INCLUDE_DIR STREQUAL "${CMAKE_OSX_SYSROOT}/usr/include")
    set_property(TARGET SQLite::SQLite3 PROPERTY INTERFACE_INCLUDE_DIRECTORIES "")
endif()

target_link_libraries(lbug_link_deps INTERFACE iceberg_duckdb SQLite::SQLite3)
install(FILES "${duckdb_runtime}" DESTINATION "${iceberg_runtime_dir}")

if(ENABLE_ICEBERG_DATAFUSION)
    find_program(CARGO cargo REQUIRED)
    set(DF_TARGET_DIR "${CMAKE_CURRENT_BINARY_DIR}/cargo" CACHE PATH "Cargo target directory for the Iceberg DataFusion planner")
    set(crate "${CMAKE_CURRENT_SOURCE_DIR}/datafusion")
    set(df_name "${CMAKE_SHARED_LIBRARY_PREFIX}lbug_iceberg_plan${CMAKE_SHARED_LIBRARY_SUFFIX}")
    set(df_lib "${DF_TARGET_DIR}/release/${df_name}")
    if(APPLE)
        set(link_args -- -C "link-arg=-Wl,-install_name,@rpath/${df_name}")
    elseif(UNIX)
        set(link_args -- -C "link-arg=-Wl,-soname,${df_name}")
    endif()
    add_custom_command(OUTPUT "${df_lib}"
        COMMAND "${CARGO}" rustc --release --locked --manifest-path "${crate}/Cargo.toml"
            --target-dir "${DF_TARGET_DIR}" ${link_args}
        DEPENDS "${crate}/Cargo.toml" "${crate}/Cargo.lock" "${crate}/src/lib.rs" VERBATIM)
    add_custom_target(iceberg_df_build DEPENDS "${df_lib}")
    add_library(iceberg_df SHARED IMPORTED GLOBAL)
    set_property(TARGET iceberg_df PROPERTY IMPORTED_LOCATION "${df_lib}")
    if(WIN32)
        set_property(TARGET iceberg_df PROPERTY IMPORTED_IMPLIB "${df_lib}.lib")
    endif()
    add_dependencies(iceberg_df iceberg_df_build)
    target_compile_definitions(lbug_iceberg_mapping PRIVATE WITH_DATAFUSION)
    target_sources(lbug_iceberg_mapping PRIVATE datafusion_plan.cpp)
    target_link_libraries(lbug_link_deps INTERFACE iceberg_df)
    install(FILES "${df_lib}" DESTINATION "${iceberg_runtime_dir}")
endif()
